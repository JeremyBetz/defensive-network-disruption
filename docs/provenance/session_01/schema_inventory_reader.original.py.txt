#!/usr/bin/env python3
import base64
import csv
import hashlib
import io
import json
import re
import subprocess
import urllib.request
from pathlib import Path

OWNER = "SkillCorner"
REPO = "opendata"
COMMIT = "02a396ffd09b283c9f092fdedeff11da6d535b66"
TREE = "44fd5081d0e6a441dbafadd12c51d6ffca8ab98b"
API = "https://api.github.com"
TOKEN = subprocess.run(["gh", "auth", "token"], check=True, capture_output=True, text=True).stdout.strip()


def request(url, *, accept="application/vnd.github+json"):
    req = urllib.request.Request(
        url,
        headers={"Accept": accept, "User-Agent": "session-1-schema-inventory", "Authorization": f"Bearer {TOKEN}"},
    )
    with urllib.request.urlopen(req, timeout=30) as response:
        return response.read()


def json_type(value):
    if value is None:
        return "null"
    if isinstance(value, bool):
        return "boolean"
    if isinstance(value, int):
        return "integer"
    if isinstance(value, float):
        return "number"
    if isinstance(value, str):
        return "string"
    if isinstance(value, list):
        return "array"
    if isinstance(value, dict):
        return "object"
    raise TypeError(type(value))


def schema(value, path="$", out=None):
    out = {} if out is None else out
    out.setdefault(path, set()).add(json_type(value))
    if isinstance(value, dict):
        for key, child in value.items():
            schema(child, f"{path}.{key}", out)
    elif isinstance(value, list):
        for child in value:
            schema(child, f"{path}[]", out)
    return out


tree_url = f"{API}/repos/{OWNER}/{REPO}/git/trees/{TREE}?recursive=1"
tree = json.loads(request(tree_url))
assert tree["truncated"] is False
entries = tree["tree"]
pattern = re.compile(
    r"^data/matches/(?P<id>\d+)/(?P=id)_(?P<product>match\.json|tracking_extrapolated\.jsonl|dynamic_events\.csv|phases_of_play\.csv)$"
)
matches = {}
for entry in entries:
    found = pattern.match(entry["path"])
    if found:
        matches.setdefault(found.group("id"), {})[found.group("product")] = {
            "path": entry["path"],
            "git_object_sha": entry["sha"],
            "git_object_size": entry.get("size"),
        }

required = {
    "match.json", "tracking_extrapolated.jsonl", "dynamic_events.csv",
    "phases_of_play.csv",
}
assert len(matches) == 20
assert all(set(products) == required for products in matches.values())

match_schemas = {}
header_sets = {"dynamic_events.csv": {}, "phases_of_play.csv": {}}
tracking_pointers = {}

for match_id in sorted(matches, key=int):
    metadata_path = matches[match_id]["match.json"]["path"]
    metadata_url = f"{API}/repos/{OWNER}/{REPO}/contents/{metadata_path}?ref={COMMIT}"
    envelope = json.loads(request(metadata_url))
    raw = base64.b64decode(envelope["content"])
    assert hashlib.sha1(b"blob " + str(len(raw)).encode() + b"\0" + raw).hexdigest() == envelope["sha"]
    parsed = json.loads(raw)
    found_schema = schema(parsed)
    normalized = {key: sorted(types) for key, types in sorted(found_schema.items())}
    schema_hash = hashlib.sha256(json.dumps(normalized, sort_keys=True).encode()).hexdigest()
    match_schemas[match_id] = {"schema_sha256": schema_hash, "field_paths": normalized}

    for product in ("dynamic_events.csv", "phases_of_play.csv"):
        path = matches[match_id][product]["path"]
        url = f"https://raw.githubusercontent.com/{OWNER}/{REPO}/{COMMIT}/{path}"
        req = urllib.request.Request(url, headers={"User-Agent": "session-1-schema-inventory", "Range": "bytes=0-65535"})
        with urllib.request.urlopen(req, timeout=30) as response:
            chunk = response.read(65536)
        first_line, separator, remainder = chunk.partition(b"\n")
        assert separator == b"\n", f"header unavailable: {path}"
        assert not first_line.endswith(b"\r\n")
        columns = next(csv.reader(io.StringIO(first_line.decode("utf-8-sig").rstrip("\r"))))
        assert columns and all(columns) and len(columns) == len(set(columns))
        header_hash = hashlib.sha256(first_line.rstrip(b"\r") + b"\n").hexdigest()
        header_sets[product][match_id] = {"header_sha256": header_hash, "columns": columns}

    pointer_path = matches[match_id]["tracking_extrapolated.jsonl"]["path"]
    pointer_url = f"{API}/repos/{OWNER}/{REPO}/git/blobs/{matches[match_id]['tracking_extrapolated.jsonl']['git_object_sha']}"
    pointer_envelope = json.loads(request(pointer_url))
    pointer = base64.b64decode(pointer_envelope["content"]).decode("ascii")
    result = re.fullmatch(
        r"version https://git-lfs\.github\.com/spec/v1\n"
        r"oid sha256:([0-9a-f]{64})\nsize ([0-9]+)\n?",
        pointer,
    )
    assert result, f"invalid LFS pointer: {pointer_path}"
    tracking_pointers[match_id] = {
        "lfs_payload_sha256": result.group(1),
        "lfs_declared_payload_size": int(result.group(2)),
        "payload_integrity": "unverified_not_downloaded",
        "payload_schema": "unverified_not_downloaded",
    }

for match_id, pointer in tracking_pointers.items():
    matches[match_id]["tracking_extrapolated.jsonl"].update(pointer)

result = {
    "schema_version": "1.0.0",
    "source": {"repository": f"https://github.com/{OWNER}/{REPO}", "commit": COMMIT, "tree": TREE},
    "checks": {
        "tree_untruncated": True,
        "unique_match_ids": len(matches),
        "complete_product_sets": True,
        "metadata_git_blob_hashes_verified": True,
        "csv_headers_only": True,
        "tracking_payloads_downloaded": False,
    },
    "matches": matches,
    "match_metadata_schemas": match_schemas,
    "csv_headers": header_sets,
}
rendered = json.dumps(result, indent=2, sort_keys=True) + "\n"
Path("/tmp/session1_schema_inventory_result.json").write_text(rendered, encoding="utf-8")
print(json.dumps({"status": "ok", "matches": len(matches), "output_sha256": hashlib.sha256(rendered.encode()).hexdigest()}))
